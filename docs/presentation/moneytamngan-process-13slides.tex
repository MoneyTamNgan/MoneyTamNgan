\begin{frame}{How TOR Data Moves Through the System}
  \begin{enumerate}\setlength\itemsep{2pt}
    \item \textbf{Discover} --- collect project metadata from the e-GP portal; classify software projects.
    \item \textbf{Download and extract} --- fetch documents; Poppler (\texttt{pdftotext}) for embedded text, Tesseract OCR for scanned pages.
    \item \textbf{Summarize and store} --- Vertex AI (Gemini) extracts structured details; MongoDB stores projects and summaries. Demo: no Vertex billing, so Gemini runs through the \texttt{agy} bridge as an automatic fallback when Vertex refuses.
    \item \textbf{Display} --- Next.js dashboard reads lists and details through the API.
  \end{enumerate}
  \shot{Insert simplified flow diagram from docs/tor-flow.md: e-GP $\to$ download $\to$ text/OCR $\to$ MongoDB $\to$ API $\to$ dashboard}
  \verified{\texttt{docs/tor-flow.md}; \texttt{lib/text-extraction.js} (\texttt{pdftotext}, \texttt{pdftoppm}); \texttt{lib/vertex/tor-extractor.js} (fallback after Vertex refuses, \texttt{AI\_FALLBACK\_PROVIDER}); \texttt{docs/agy-bridge.md}. Note: the code names the protocol \texttt{ollama}; \texttt{agy} is an alias (\texttt{.env.example}).}
  \note{We collect metadata, download the source document, extract text, use OCR for scans, and store the result so the dashboard can fetch it through the API.}
\end{frame}
